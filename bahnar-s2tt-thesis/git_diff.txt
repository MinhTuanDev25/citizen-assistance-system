+def filter_by_duration(
+    df: pd.DataFrame,
+    *,
+    min_duration: float = FULL_PREPARE_MIN_DURATION,
+    max_duration: float = FULL_PREPARE_MAX_DURATION,
+    duration_col: str = "duration_seconds",
+) -> Tuple[pd.DataFrame, pd.DataFrame]:
+    """Return (kept, excluded) by inclusive duration gate on metadata."""
+    if df is None or len(df) == 0:
+        empty = df.iloc[0:0].copy() if df is not None else pd.DataFrame()
+        return empty, empty
+    durs = pd.to_numeric(df[duration_col], errors="coerce")
+    ok = durs.notna() & np.isfinite(durs.to_numpy(dtype=float, copy=False))
+    ok &= (durs >= float(min_duration)) & (durs <= float(max_duration))
+    kept = df.loc[ok].copy().reset_index(drop=True)
+    excl = df.loc[~ok].copy().reset_index(drop=True)
+    return kept, excl

+def filter_nonempty_normalized_text(
+    df: pd.DataFrame,
+    *,
+    text_col: str = "text_bahnar",
+    normalize_fn: Optional[Callable[[Any], str]] = None,
+) -> Tuple[pd.DataFrame, pd.DataFrame]:
+    """Return (kept, excluded) where kept has non-empty normalized transcript."""
+    if normalize_fn is None:
+        normalize_fn = normalize_bahnar_ctc_v1
+    if df is None or len(df) == 0:
+        empty = df.iloc[0:0].copy() if df is not None else pd.DataFrame()
+        return empty, empty
+    norms = [normalize_fn(t) for t in df[text_col].tolist()]
+    ok = [(n is not None and str(n).strip() != "") for n in norms]
+    out = df.copy()
+    out["text_bahnar_norm"] = norms
+    kept = out.loc[ok].copy().reset_index(drop=True)
+    excl = out.loc[[not x for x in ok]].copy().reset_index(drop=True)
+    return kept, excl

+def check_split_overlaps(
+    train_df: pd.DataFrame,
+    val_df: pd.DataFrame,
+    cols: Sequence[str] = ("record_uid", "group_id", "recording_group_id", "pair_key"),
+) -> Dict[str, int]:
+    """Count overlaps between train/validation for each key column."""
+    overlaps: Dict[str, int] = {}
+    for col in cols:
+        if col not in train_df.columns or col not in val_df.columns:
+            overlaps[col] = -1
+            continue
+        a = set(train_df[col].dropna().astype(str))
+        b = set(val_df[col].dropna().astype(str))
+        overlaps[col] = len(a & b)
+    return overlaps

+def group_records_by_parquet(df: pd.DataFrame) -> List[Tuple[str, pd.DataFrame]]:
+    """
+    Deterministic shard order: sorted parquet_file, rows within shard sorted by
+    shard_row_index then record_uid.
+    """
+    if len(df) == 0:
+        return []
+    work = df.copy()
+    work["_shard_row"] = pd.to_numeric(work["shard_row_index"], errors="coerce").fillna(-1).astype(int)
+    groups: List[Tuple[str, pd.DataFrame]] = []
+    for pq_name in sorted(work["parquet_file"].astype(str).unique().tolist()):
+        sub = work.loc[work["parquet_file"].astype(str) == pq_name].copy()
+        sub = sub.sort_values(["_shard_row", "record_uid"]).reset_index(drop=True)
+        groups.append((pq_name, sub.drop(columns=["_shard_row"])))
+    return groups

+@dataclass
+class ShardReadTracker:
+    """Ensures each parquet shard is opened for sequential batch reading at most once."""
+
+    opened: Set[str] = field(default_factory=set)
+    open_counts: Dict[str, int] = field(default_factory=dict)
+
+    def mark_open(self, parquet_file: str) -> None:
+        key = str(parquet_file)
+        self.open_counts[key] = self.open_counts.get(key, 0) + 1
+        if key in self.opened:
+            raise RuntimeError(
+                f"Parquet shard opened more than once (must be sequential single-pass): {key}"
+            )
+        self.opened.add(key)

+def iter_parquet_batches_pyarrow(
+    local_parquet_path: Path,
+    *,
+    columns: Sequence[str],
+    batch_size: int = 64,
+) -> Iterator[pd.DataFrame]:
+    """
+    Sequential batch iterator over a local parquet file.
+
+    Uses ParquetFile.iter_batches — never pq.read_table() of the full audio column.
+    """
+    import pyarrow.parquet as pq
+
+    pf = pq.ParquetFile(str(local_parquet_path))
+    # Only request columns that exist
+    available = set(pf.schema_arrow.names)
+    cols = [c for c in columns if c in available]
+    for batch in pf.iter_batches(batch_size=int(batch_size), columns=cols):
+        yield batch.to_pandas()

+def evaluate_record_ctc(
+    text_bahnar: Any,
+    duration_sec: float,
+    *,
+    vocab: Dict[str, int],
+    target_sr: int,
+    normalize_fn: Optional[Callable[[Any], str]] = None,
+    encode_fn: Optional[Callable[..., List[int]]] = None,
+) -> Dict[str, Any]:
+    if normalize_fn is None:
+        normalize_fn = normalize_bahnar_ctc_v1
+    if encode_fn is None:
+        encode_fn = encode_text_with_vocab
+    norm = normalize_fn(text_bahnar)
+    if norm is None or str(norm).strip() == "":
+        return {"ok": False, "reason": "empty_normalized_reference", "norm": ""}
+    tokens = encode_fn(norm, vocab)
+    feas = check_ctc_feasibility(tokens, int(float(duration_sec) * int(target_sr)))
+    if not feas.get("feasible"):
+        return {"ok": False, "reason": feas.get("reason") or "ctc_infeasible", "norm": norm, "feas": feas}
+    return {"ok": True, "reason": "ok", "norm": norm, "feas": feas}

+def process_shard_records(
+    shard_name: str,
+    shard_df: pd.DataFrame,
+    *,
+    split: str,
+    vocab: Dict[str, int],
+    dataset_revision: str,
+    target_sr: int,
+    min_duration: float,
+    max_duration: float,
+    expected_processing_version: str,
+    nb02_cache_dir: Path,
+    batch_reader: Callable[[str, Sequence[int]], Dict[int, Dict[str, Any]]],
+    tracker: ShardReadTracker,
+    decode_and_qa: Optional[Callable[[Any], Dict[str, Any]]] = None,
+    local_audio_cache_dir: Optional[Path] = None,
+    write_wav_fn: Optional[Callable[[Path, Any, int], None]] = None,
+) -> Dict[str, Any]:
+    """
+    Process one parquet shard's candidate rows.
+
+    ``batch_reader(shard_name, row_indices)`` must return a mapping
+    shard_row_index -> {"audio": ..., "id": ...} for those rows, reading the
+    shard sequentially in batches (tests inject a fake reader).
+
+    When ``local_audio_cache_dir`` is set, decoded eligible audio is written as
+    PCM16 WAV on *local* disk only (never under Google Drive FULL_STATE_DIR).
+    ``decode_and_qa`` must then return ``waveform`` (16 kHz mono float) on success.
+    """
+    tracker.mark_open(shard_name)
+    needed_idxs = sorted(
+        int(x) for x in pd.to_numeric(shard_df["shard_row_index"], errors="coerce").dropna().tolist()
+    )
+    audio_map = batch_reader(shard_name, needed_idxs) if needed_idxs else {}
+
+    eligible_rows: List[Dict[str, Any]] = []
+    exclusions: List[Dict[str, Any]] = []
+    local_dir = Path(local_audio_cache_dir) if local_audio_cache_dir is not None else None
+    if local_dir is not None:
+        local_dir.mkdir(parents=True, exist_ok=True)
+    _write_wav = write_wav_fn
+    if local_dir is not None and _write_wav is None:
+        from src.data_utils import write_pcm16_wav as _write_wav
+
+    for _, row in shard_df.iterrows():
+        uid = str(row["record_uid"])
+        rid = str(row["record_id"])
+        try:
+            srow = int(row["shard_row_index"])
+        except (TypeError, ValueError):
+            exclusions.append(exclusion_row(split=split, row=row, reason="bad_shard_row_index"))
+            continue
+
+        # Prefer NB02 cache reuse (no Drive writes)
+        reused = try_reuse_nb02_cache(
+            uid,
+            nb02_cache_dir=nb02_cache_dir,
+            dataset_revision=dataset_revision,
+            target_sr=target_sr,
+            min_duration=min_duration,
+            max_duration=max_duration,
+            expected_processing_version=expected_processing_version,
+        )
+        if reused is not None:
+            dur = float(reused["processed_duration_seconds"] or row.get("duration_seconds") or 0.0)
+            ctc = evaluate_record_ctc(row.get("text_bahnar"), dur, vocab=vocab, target_sr=target_sr)
+            if not ctc["ok"]:
+                exclusions.append(exclusion_row(split=split, row=row, reason=str(ctc["reason"])))
+                continue
+            eligible_rows.append({
+                **{c: row.get(c) for c in [
+                    "record_uid", "record_id", "group_id", "recording_group_id", "pair_key",
+                    "source_split", "split", "parquet_file", "shard_row_index", "text_bahnar",
+                    "duration_seconds",
+                ]},
+                "text_bahnar_norm": ctc["norm"],
+                "processed_duration_seconds": dur,
+                "local_cache_relpath": reused["local_cache_relpath"],
+                "audio_source": reused["audio_source"],
+            })
+            continue
+
+        # Need audio from parquet batch map
+        payload = audio_map.get(srow)
+        if payload is None:
+            exclusions.append(exclusion_row(split=split, row=row, reason="audio_row_missing_in_shard"))
+            continue
+
+        # Identity check
+        payload_id = payload.get("id")
+        if payload_id is not None and str(payload_id) != rid:
+            exclusions.append(exclusion_row(
+                split=split, row=row, reason="record_id_mismatch",
+                detail=f"expected={rid} got={payload_id}",
+            ))
+            continue
+
+        if decode_and_qa is None:
+            exclusions.append(exclusion_row(split=split, row=row, reason="decode_fn_missing"))
+            continue
+
+        qa = decode_and_qa(payload.get("audio"))
+        if not qa.get("ok"):
+            exclusions.append(exclusion_row(
+                split=split, row=row, reason=f"audio_qa_{qa.get('reason') or 'fail'}",
+            ))
+            continue
+        dur = float(qa.get("duration_sec") or 0.0)
+        if not (min_duration <= dur <= max_duration):
+            exclusions.append(exclusion_row(split=split, row=row, reason="decoded_duration_out_of_range"))
+            continue
+
+        ctc = evaluate_record_ctc(row.get("text_bahnar"), dur, vocab=vocab, target_sr=target_sr)
+        if not ctc["ok"]:
+            exclusions.append(exclusion_row(split=split, row=row, reason=str(ctc["reason"])))
+            continue
+
+        rel = safe_cache_filename(uid)
+        if local_dir is not None:
+            wav = qa.get("waveform")
+            if wav is None:
+                exclusions.append(exclusion_row(
+                    split=split, row=row, reason="missing_waveform_for_local_cache",
+                ))
+                continue
+            out_path = local_dir / rel
+            try:
+                _write_wav(out_path, wav, int(target_sr))
+            except Exception as exc:
+                exclusions.append(exclusion_row(
+                    split=split, row=row, reason="local_wav_write_failed",
+                    detail=f"{type(exc).__name__}: {exc}",
+                ))
+                continue
+
+        # Local-only cache key (never bulk WAV on Drive)
+        eligible_rows.append({
+            **{c: row.get(c) for c in [
+                "record_uid", "record_id", "group_id", "recording_group_id", "pair_key",
+                "source_split", "split", "parquet_file", "shard_row_index", "text_bahnar",
+                "duration_seconds",
+            ]},
+            "text_bahnar_norm": ctc["norm"],
+            "processed_duration_seconds": dur,
+            "local_cache_relpath": rel,
+            "audio_source": "prepared_local",
+        })
+
+    return {"eligible": eligible_rows, "exclusions": exclusions}
+

+def compute_steps_per_epoch(
+    n_train: int,
+    *,
+    per_device_train_batch_size: int = 1,
+    gradient_accumulation_steps: int = 8,
+    n_devices: int = 1,
+) -> int:
+    """HF-style steps per epoch from eligible train size and effective batch."""
+    if n_train <= 0:
+        raise ValueError("n_train must be positive")
+    eff = max(1, int(per_device_train_batch_size) * int(gradient_accumulation_steps) * int(n_devices))
+    return int(math.ceil(int(n_train) / eff))

+def atomic_copy_tree(src: Union[str, Path], dst: Union[str, Path]) -> Path:
+    """Copy directory to dst via temporary sibling then os.replace."""
+    src = Path(src)
+    dst = Path(dst)
+    if not src.exists():
+        raise FileNotFoundError(src)
+    dst.parent.mkdir(parents=True, exist_ok=True)
+    tmp = dst.parent / (dst.name + ".tmp_copy")
+    if tmp.exists():
+        shutil.rmtree(tmp)
+    if src.is_dir():
+        shutil.copytree(src, tmp)
+    else:
+        shutil.copy2(src, tmp)
+    if dst.exists():
+        if dst.is_dir():
+            shutil.rmtree(dst)
+        else:
+            dst.unlink()
+    os.replace(str(tmp), str(dst))
+    return dst

+    """
+    Inspect HF Trainer checkpoint artifacts for resume invariants.
+    Does not load tensors — presence + trainer_state.json fields only.

+    global_step = int(state.get("global_step") or 0)

+        return any((cp / n).exists() for n in names)
+
+    model_ok = _exists(
+        "pytorch_model.bin",
+        "model.safetensors",
+        "adapter_model.bin",
+        "adapter_model.safetensors",
+    )
+    # Also accept sharded / HF weight index
+    if not model_ok:
+        model_ok = any(cp.glob("pytorch_model*.bin")) or any(cp.glob("model*.safetensors"))
+    optimizer_ok = _exists("optimizer.pt", "optimizer.bin")
+    scheduler_ok = _exists("scheduler.pt", "scheduler.bin")
+    rng_ok = _exists("rng_state.pth", "rng_state_0.pth")

+        "path": str(cp),
+        "global_step": global_step,
+        "model_ok": model_ok,

+        "scheduler_ok": scheduler_ok,

+def select_best_checkpoint_by_cer(
+    history_rows: Sequence[Dict[str, Any]],
+) -> Optional[Dict[str, Any]]:
+    """Pick checkpoint entry with lowest CER from eval history rows."""
+    best = None
+    for row in history_rows:
+        cer = row.get("cer")
+        if cer is None:
+            continue
+        try:
+            cer_f = float(cer)
+        except (TypeError, ValueError):
+            continue
+        if best is None or cer_f < float(best["cer"]):
+            best = dict(row)
+            best["cer"] = cer_f
+    return best
